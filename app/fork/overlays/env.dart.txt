import 'package:omi/fork/identity/deployment.dart';

/// Selected fork profile is the only routing authority in this artifact.
abstract class Env {
  static NativeDeployment get profile => NativeDeployment.current;
  static String? get apiBaseUrl => profile.apiBase;
  static bool isTestFlight = false;
  static String? get posthogApiKey => null;
  static String? get googleMapsApiKey => null;
  static String? get intercomAppId => null;
  static String? get intercomIOSApiKey => null;
  static String? get intercomAndroidApiKey => null;
  static String? get googleClientId => null;
  static String? get googleClientSecret => null;
  static bool get useWebAuth => false;
  static bool get useAuthCustomToken => false;
}

// Build-runner still visits the original, unused envied declarations. These
// compile-time fields cannot override the selected runtime topology above.
abstract class EnvFields {
  String? get posthogApiKey;
  String? get apiBaseUrl;
  String? get googleMapsApiKey;
  String? get intercomAppId;
  String? get intercomIOSApiKey;
  String? get intercomAndroidApiKey;
  String? get googleClientId;
  String? get googleClientSecret;
  bool? get useWebAuth;
  bool? get useAuthCustomToken;
}
